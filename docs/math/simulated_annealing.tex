\begin{abstract}
This document outlines the mathematical foundation of the Simulated Annealing (SA) metaheuristic applied to the Rolling Stock Scheduling and Circulation Problem. We formally define the solution state space, the Metropolis acceptance probability $P(\text{accept}) = e^{-\Delta/T}$, the geometric cooling schedule $T_k = T_0 \cdot r^k$, and the mathematical operators for Relocate, Swap, and Couple/Decouple neighborhood transitions. Finally, we provide a theoretical note on the convergence limitations of classical SA on large-scale instances.
\end{abstract}

\section{Introduction and Analogy}
Simulated Annealing is a stochastic local search metaheuristic inspired by the physical annealing process in metallurgy, wherein a material is heated and slowly cooled to achieve a low-energy crystalline structure. In rolling stock circulation, the objective cost $\mathcal{J}(\mathcal{S})$ corresponds to physical energy, and the temperature parameter $T$ controls the probability of accepting non-improving schedule perturbations.

\section{State Representation}
Let $\mathcal{S}$ denote a candidate circulation schedule represented as an indexed partition of trips assigned to trainset units:
\begin{equation}
    \mathcal{S} = \{ \mathcal{C}_u \mid u \in U \}
\end{equation}
where each chain $\mathcal{C}_u = (i_1, i_2, \dots, i_{k_u})$ is an ordered sequence of timetable trips assigned to unit $u$, sorted chronologically by scheduled departure time:
\begin{equation}
    \text{dep}_{i_1} \le \text{dep}_{i_2} \le \dots \le \text{dep}_{i_{k_u}}
\end{equation}

\section{Formal Definition of Neighborhood Moves}
A neighboring solution $\mathcal{S}' \in \mathcal{N}(\mathcal{S})$ is generated by applying one of three stochastic neighborhood operators:

\subsection{1. Relocate Operator ($\mathcal{M}_{\text{relocate}}$)}
The relocate operator transfers a single trip from a source unit to a destination unit:
\begin{equation}
    \mathcal{M}_{\text{relocate}}(\mathcal{S}, u_{\text{src}}, i, u_{\text{dst}}) = \mathcal{S}'
\end{equation}
where:
\begin{align}
    \mathcal{C}'_{u_{\text{src}}} &= \mathcal{C}_{u_{\text{src}}} \setminus \{i\} \\
    \mathcal{C}'_{u_{\text{dst}}} &= \text{SortByDeparture}\left( \mathcal{C}_{u_{\text{dst}}} \cup \{i\} \right) \\
    \mathcal{C}'_u &= \mathcal{C}_u, \quad \forall u \in U \setminus \{u_{\text{src}}, u_{\text{dst}}\}
\end{align}
To preserve feasibility, candidate target insertions are filtered by verifying the spatio-temporal arc conditions:
\begin{equation}
    \text{arr}_{i_{\text{prev}}} + \tau_{i_{\text{prev}}} + t(d_{i_{\text{prev}}}, o_i) \le \text{dep}_i \quad \text{and} \quad \text{arr}_i + \tau_i + t(d_i, o_{i_{\text{next}}}) \le \text{dep}_{i_{\text{next}}}
\end{equation}

\subsection{2. Swap Operator ($\mathcal{M}_{\text{swap}}$)}
The swap operator exchanges two trips assigned to different active units:
\begin{equation}
    \mathcal{M}_{\text{swap}}(\mathcal{S}, u_1, i_1, u_2, i_2) = \mathcal{S}'
\end{equation}
where $i_1 \in \mathcal{C}_{u_1}$ and $i_2 \in \mathcal{C}_{u_2}$ such that:
\begin{align}
    \mathcal{C}'_{u_1} &= \text{SortByDeparture}\left( (\mathcal{C}_{u_1} \setminus \{i_1\}) \cup \{i_2\} \right) \\
    \mathcal{C}'_{u_2} &= \text{SortByDeparture}\left( (\mathcal{C}_{u_2} \setminus \{i_2\}) \cup \{i_1\} \right) \\
    \mathcal{C}'_u &= \mathcal{C}_u, \quad \forall u \in U \setminus \{u_1, u_2\}
\end{align}

\subsection{3. Couple / Decouple Operator ($\mathcal{M}_{\text{couple}}$)}
The coupling operator alters the multi-unit consist configuration of a trip to match passenger demand $\rho_i$:
\begin{equation}
    \mathcal{M}_{\text{couple}}(\mathcal{S}, i, u_{\text{cand}}) = \mathcal{S}'
\end{equation}
\begin{itemize}
    \item \textbf{Couple}: If $\sum_{u: i \in \mathcal{C}_u} c_u < \rho_i$, an unassigned or temporally idle unit $u_{\text{cand}}$ is coupled to trip $i$:
    \begin{equation}
        \mathcal{C}'_{u_{\text{cand}}} = \text{SortByDeparture}\left(\mathcal{C}_{u_{\text{cand}}} \cup \{i\}\right)
    \end{equation}
    \item \textbf{Decouple}: If the aggregate capacity significantly exceeds demand $\sum_{u: i \in \mathcal{C}_u} c_u \ge \rho_i + c_{u_{\text{rem}}}$, a redundant unit $u_{\text{rem}}$ is decoupled:
    \begin{equation}
        \mathcal{C}'_{u_{\text{rem}}} = \mathcal{C}_{u_{\text{rem}}} \setminus \{i\}
    \end{equation}
\end{itemize}

\section{Metropolis Acceptance Criterion}
Let $\mathcal{S}$ be the current state and $\mathcal{S}'$ a proposed neighboring state. The change in the scalarized objective cost is:
\begin{equation}
    \Delta = \mathcal{J}(\mathcal{S}') - \mathcal{J}(\mathcal{S})
\end{equation}
The probability of accepting state transition $\mathcal{S} \to \mathcal{S}'$ at temperature $T$ is governed by the Metropolis acceptance criterion:
\begin{equation}
    P(\text{accept}) = 
    \begin{cases}
        1, & \text{if } \Delta \le 0 \\
        \exp\left(-\frac{\Delta}{T}\right), & \text{if } \Delta > 0
    \end{cases}
\end{equation}
If $\Delta > 0$, a random variable $\xi \sim \mathcal{U}(0, 1)$ is drawn; the transition is accepted if and only if $\xi < \exp(-\Delta / T)$.

\section{Geometric Cooling Schedule}
The temperature decreases across successive outer cooling epochs $k = 0, 1, 2, \dots$ according to a geometric schedule:
\begin{equation}
    T_k = T_0 \cdot r^k
\end{equation}
where:
\begin{itemize}
    \item $T_0 = 500.0$: Initial high temperature enabling broad exploration of the solution space.
    \item $r = 0.995$: Cooling rate parameter ($0 < r < 1$).
    \item $T_{\min} = 0.5$: Minimum stopping temperature threshold.
    \item $L = 40$: Number of inner Metropolis iterations evaluated at each constant temperature step $T_k$.
\end{itemize}

\section{Algorithmic Structure}
Algorithm~\ref{alg:sa} details the complete Simulated Annealing optimization procedure.

\begin{algorithm}[H]
\caption{Simulated Annealing for Rolling Stock Scheduling}
\label{alg:sa}
\begin{algorithmic}[1]
\Require Initial greedy solution $\mathcal{S}_0$, Parameters $T_0, r, T_{\min}, L$
\Ensure Best discovered schedule $\mathcal{S}^*$, Cost $\mathcal{J}^*$
\State $\mathcal{S} \gets \mathcal{S}_0$, $\mathcal{S}^* \gets \mathcal{S}_0$
\State $\mathcal{J} \gets \text{ComputeCost}(\mathcal{S})$, $\mathcal{J}^* \gets \mathcal{J}$
\State $T \gets T_0$
\While{$T > T_{\min}$}
    \For{$\text{iter} = 1$ \textbf{to} $L$}
        \State Sample move $\mathcal{M} \in \{\mathcal{M}_{\text{relocate}}, \mathcal{M}_{\text{swap}}, \mathcal{M}_{\text{couple}}\}$
        \State $\mathcal{S}' \gets \mathcal{M}(\mathcal{S})$
        \State $\mathcal{J}' \gets \text{ComputeCost}(\mathcal{S}')$
        \State $\Delta \gets \mathcal{J}' - \mathcal{J}$
        \If{$\Delta \le 0$ \textbf{or} $\text{rand}() < \exp(-\Delta / T)$}
            \State $\mathcal{S} \gets \mathcal{S}'$, $\mathcal{J} \gets \mathcal{J}'$
            \If{$\mathcal{J} < \mathcal{J}^*$}
                \State $\mathcal{S}^* \gets \mathcal{S}$, $\mathcal{J}^* \gets \mathcal{J}$
            \EndIf
        \EndIf
    \EndFor
    \State $T \gets r \cdot T$
\EndWhile
\State \Return $\mathcal{S}^*, \mathcal{J}^*$
\end{algorithmic}
\end{algorithm}

\section{Convergence Limitations on Large Instances}
While Simulated Annealing theoretically guarantees asymptotic convergence to the global optimum as $T \to 0$, practical implementations face severe computational barriers on large-scale rolling stock instances:
\begin{enumerate}
    \item \textbf{Cooling Rate vs. Markov Chain Length}: Theoretical convergence requires a logarithmic cooling schedule $T_k = \frac{c}{\ln(k+1)}$, which requires impractically large cooling epochs ($k \to \infty$). Practical geometric cooling ($r = 0.995$) trades convergence guarantees for computational tractability.
    \item \textbf{Exponential State Space Growth}: As the number of trips $|T|$ and fleet size $|U|$ increase, the size of the combinatorial circulation space grows as $\mathcal{O}(|U|^{|T|})$.
    \item \textbf{Trapping in Local Optima}: In dense timetable instances with tight turnaround buffers and maintenance thresholds, feasible schedules occupy narrow, disconnected valleys within the energy landscape. Once the temperature drops below $T \approx 10$, the probability of accepting uphill barrier transitions ($\exp(-\Delta / T) \approx 0$) vanishes. As a result, the Markov chain becomes trapped in local optima, unable to escape without extensive multi-step rearrangements.
\end{enumerate}
