\section{Rolling Stock Scheduling: Mathematical Problem Formulation}

\subsection{Sets and Indices}
\begin{itemize}
    \item $U$: Set of rolling stock trainset units, indexed by $u \in U$.
    \item $T$: Set of revenue passenger trips, indexed by $i, j \in T$.
    \item $S$: Set of stations in the railway network, indexed by $s \in S$.
    \item $D \subseteq S$: Subset of stations designated as depot facilities.
    \item $D_{\text{maint}} \subseteq D$: Subset of depots equipped with maintenance workshops.
\end{itemize}

\subsection{Parameters}
\begin{itemize}
    \item Trip parameters for each $i \in T$:
    \begin{itemize}
        \item $o_i, d_i \in S$: Origin and destination stations.
        \item $\text{dep}_i, \text{arr}_i \in \mathbb{R}_+$: Scheduled departure and arrival times (minutes from midnight).
        \item $\tau_i \in \mathbb{R}_+$: Minimum required turnaround time at destination $d_i$.
        \item $\ell_i \in \mathbb{R}_+$: Scheduled track distance of trip $i$ (km).
        \item $\rho_i \in \mathbb{Z}_+$: Peak/expected passenger demand count for trip $i$.
    \end{itemize}
    \item Unit parameters for each $u \in U$:
    \begin{itemize}
        \item $c_u \in \mathbb{Z}_+$: Nominal seated passenger capacity of unit $u$.
        \item $h_u \in D$: Initial stabling home depot of unit $u$.
        \item $a_u \in \mathbb{R}_+$: Earliest availability timestamp of unit $u$ at depot $h_u$.
        \item $\mu_u \in \mathbb{R}_+$: Mileage accumulated since last overhaul inspection (km).
        \item $M_{\text{type}(u)} \in \mathbb{R}_+$: Maximum permissible run distance prior to required maintenance.
    \end{itemize}
    \item Network topology parameters for station pairs $(s_1, s_2) \in S \times S$:
    \begin{itemize}
        \item $t(s_1, s_2) \in \mathbb{R}_+$: Deadhead transit runtime between stations (minutes).
        \item $\delta(s_1, s_2) \in \mathbb{R}_+$: Track distance between stations (km).
    \end{itemize}
\end{itemize}

\subsection{Decision Variables}
\begin{itemize}
    \item $x_{u,i} \in \{0, 1\}$: Binary variable equal to 1 if rolling stock unit $u$ is assigned to trip $i$, and 0 otherwise.
    \item $y_{u,i,j} \in \{0, 1\}$: Binary transition variable indicating unit $u$ operates trip $j$ immediately following trip $i$.
    \item $y^0_{u,i} \in \{0, 1\}$: Binary variable indicating unit $u$ departs from depot $h_u$ to serve trip $i$ as its initial assignment.
    \item $z_u \in \{0, 1\}$: Binary variable indicating if rolling stock unit $u$ is active during the operating horizon.
    \item $s_i^{\text{dem}} \ge 0$: Uncovered passenger demand slack variable for trip $i$.
    \item $v_{u,i,j}^{\text{time}} \ge 0$: Timing shortfall violation for consecutive transition $(i, j)$ on unit $u$.
\end{itemize}

\subsection{Constraints}
\paragraph{1. Passenger Demand Satisfaction}
\begin{equation}
    \sum_{u \in U} c_u x_{u,i} + s_i^{\text{dem}} \ge \rho_i, \quad \forall i \in T
\end{equation}

\paragraph{2. Flow Conservation along Circulation Chains}
\begin{equation}
    y^0_{u,i} + \sum_{j \in T \setminus \{i\}} y_{u,j,i} = x_{u,i}, \quad \forall u \in U, \forall i \in T
\end{equation}
\begin{equation}
    \sum_{j \in T \setminus \{i\}} y_{u,i,j} \le x_{u,i}, \quad \forall u \in U, \forall i \in T
\end{equation}
\begin{equation}
    \sum_{i \in T} y^0_{u,i} \le z_u, \quad \forall u \in U
\end{equation}

\paragraph{3. Unit Activation Linking}
\begin{equation}
    x_{u,i} \le z_u, \quad \forall u \in U, \forall i \in T
\end{equation}

\paragraph{4. Temporal and Spatial Feasibility}
For every consecutive transition $(i, j)$ operated by unit $u$:
\begin{equation}
    \text{arr}_i + \tau_i + t(d_i, o_j) - \text{dep}_j \le v_{u,i,j}^{\text{time}} + M (1 - y_{u,i,j}), \quad \forall u \in U, \forall i, j \in T
\end{equation}
For initial depot pull-out:
\begin{equation}
    a_u + t(h_u, o_i) - \text{dep}_i \le M (1 - y^0_{u,i}), \quad \forall u \in U, \forall i \in T
\end{equation}

\paragraph{5. Mileage Maintenance Bounding}
Let $\mathcal{C}_u$ denote the ordered sequence of trips assigned to unit $u$. Total cumulative distance must satisfy:
\begin{equation}
    \mu_u + \sum_{i \in \mathcal{C}_u} \left( \delta(d_{\text{prev}(i)}, o_i) + \ell_i \right) \le M_{\text{type}(u)} + v_u^{\text{maint}}, \quad \forall u \in U
\end{equation}

\subsection{Comprehensive Objective Function}
\begin{equation}
    \min \quad \mathcal{J} = W_{\text{deadhead}} \cdot \mathcal{D} + W_{\text{unit}} \cdot \mathcal{U} + W_{\text{maint}} \cdot \mathcal{M} + W_{\text{demand}} \cdot \mathcal{P} + W_{\text{timing}} \cdot \mathcal{V}_{\text{time}}
\end{equation}
where:
\begin{align}
    \mathcal{D} &= \sum_{u \in U} \left[ \sum_{i \in T} \delta(h_u, o_i) y^0_{u,i} + \sum_{i \in T} \sum_{j \in T \setminus \{i\}} \delta(d_i, o_j) y_{u,i,j} \right] \\
    \mathcal{U} &= \sum_{u \in U} z_u \\
    \mathcal{M} &= \sum_{u \in U} v_u^{\text{maint}} \\
    \mathcal{P} &= \sum_{i \in T} s_i^{\text{dem}} \\
    \mathcal{V}_{\text{time}} &= \sum_{u \in U} \sum_{i \in T} \sum_{j \in T} v_{u,i,j}^{\text{time}}
\end{align}
