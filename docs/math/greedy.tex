\begin{abstract}
This document details the deterministic Greedy Construction heuristic for the Rolling Stock Scheduling and Circulation Problem. We present the formal algorithmic procedure in pseudocode using \texttt{algorithmicx}, analyze its asymptotic time complexity of $\mathcal{O}(|T| \cdot |U|)$, and theoretically examine why greedy dispatch fails due to myopic decision-making and lack of downstream look-ahead.
\end{abstract}

\section{Introduction}
The greedy construction algorithm establishes a deterministic, sequential baseline for the rolling stock circulation problem. Timetable trips are processed strictly in chronological order of scheduled departure:
\begin{equation}
    T = (i_1, i_2, \dots, i_{|T|}), \quad \text{such that } \text{dep}_{i_1} \le \text{dep}_{i_2} \le \dots \le \text{dep}_{i_{|T|}}
\end{equation}
At each step, the algorithm assigns trainset units to the current trip based on immediate temporal availability and minimum deadhead repositioning distance.

\section{Unit State Representation}
For each rolling stock unit $u \in U$, a dynamic operational state is maintained:
\begin{equation}
    \sigma_u = \left( \text{loc}_u, \, \text{avail}_u, \, \mu_u \right)
\end{equation}
initialized at $t = 0$ as:
\begin{align}
    \text{loc}_u^{(0)} &= h_u \quad &&\text{(initial home depot)} \\
    \text{avail}_u^{(0)} &= a_u \quad &&\text{(earliest departure time)} \\
    \mu_u^{(0)} &= \mu_u^{\text{init}} \quad &&\text{(initial accumulated mileage)}
\end{align}

\section{Algorithmic Procedure}
Algorithm~\ref{alg:greedy} presents the complete pseudocode for the greedy construction heuristic.

\begin{algorithm}[H]
\caption{Greedy Construction Heuristic for Rolling Stock Circulation}
\label{alg:greedy}
\begin{algorithmic}[1]
\Require Fleet $U$, Timetable trips $T$, Distance matrix $\delta$, Travel time matrix $t$, Demands $\rho$
\Ensure Circulation chains $\mathcal{C}_u$ for all $u \in U$, Schedule $\mathcal{S}$
\State Sort trips $T = (i_1, i_2, \dots, i_{|T|})$ such that $\text{dep}_{i_1} \le \text{dep}_{i_2} \le \dots \le \text{dep}_{i_{|T|}}$
\For{each unit $u \in U$}
    \State $\mathcal{C}_u \gets \emptyset$, $\text{loc}_u \gets h_u$, $\text{avail}_u \gets a_u$, $\mu_u \gets \mu_u^{\text{init}}$
\EndFor
\For{each trip $i \in T$}
    \State $\text{unmet\_demand} \gets \rho_i$
    \State $\mathcal{U}_{\text{feas}} \gets \emptyset$
    \For{each unit $u \in U$}
        \State $\text{transit\_time} \gets t(\text{loc}_u, o_i)$
        \If{$\text{avail}_u + \text{transit\_time} \le \text{dep}_i$}
            \State $\text{dist} \gets \delta(\text{loc}_u, o_i)$
            \State Add $(u, \text{dist})$ to $\mathcal{U}_{\text{feas}}$
        \EndIf
    \EndFor
    \If{$\mathcal{U}_{\text{feas}} \ne \emptyset$}
        \State Sort $\mathcal{U}_{\text{feas}}$ in ascending order of deadhead distance $\text{dist}$
        \State Select best primary unit $u^* \gets \arg\min_{(u, d) \in \mathcal{U}_{\text{feas}}} d$
        \State Assign $u^*$ to trip $i$: $\mathcal{C}_{u^*} \gets \mathcal{C}_{u^*} \cup \{i\}$
        \State Update: $\text{loc}_{u^*} \gets d_i$, $\text{avail}_{u^*} \gets \text{arr}_i + \tau_i$, $\mu_{u^*} \gets \mu_{u^*} + \delta(\text{loc}_{u^*}, o_i) + \ell_i$
        \State $\text{unmet\_demand} \gets \max(0, \text{unmet\_demand} - c_{u^*})$
        \If{$\text{unmet\_demand} > 0$ and $\mathcal{U}_{\text{feas}} \setminus \{u^*\} \ne \emptyset$}
            \State Select secondary unit $u_2^* \gets \arg\min_{(u, d) \in \mathcal{U}_{\text{feas}} \setminus \{u^*\}} d$
            \State Assign $u_2^*$ to trip $i$: $\mathcal{C}_{u_2^*} \gets \mathcal{C}_{u_2^*} \cup \{i\}$
            \State Update: $\text{loc}_{u_2^*} \gets d_i$, $\text{avail}_{u_2^*} \gets \text{arr}_i + \tau_i$, $\mu_{u_2^*} \gets \mu_{u_2^*} + \delta(\text{loc}_{u_2^*}, o_i) + \ell_i$
        \EndIf
    \Else
        \State \Comment{Fallback: assign earliest available unit despite timing violation}
        \State $u_{\text{fallback}} \gets \arg\min_{u \in U} \text{avail}_u$
        \State Assign $u_{\text{fallback}}$ to trip $i$: $\mathcal{C}_{u_{\text{fallback}}} \gets \mathcal{C}_{u_{\text{fallback}}} \cup \{i\}$
        \State Update state of $u_{\text{fallback}}$ accordingly
    \EndIf
\EndFor
\State \Return circulation chains $\{ \mathcal{C}_u \mid u \in U \}$
\end{algorithmic}
\end{algorithm}

\section{Complexity Analysis}
Let $|T|$ denote the total number of timetable trips and $|U|$ the total fleet size.
\begin{enumerate}
    \item \textbf{Initial Sorting}: Sorting timetable trips chronologically by departure timestamp $\text{dep}_i$ takes $\mathcal{O}(|T| \log |T|)$ operations.
    \item \textbf{Per-Trip Evaluation Loop}: For each trip $i \in \{1, \dots, |T|\}$:
    \begin{itemize}
        \item Evaluating temporal availability and repositioning distance across all candidate units takes $\mathcal{O}(|U|)$ checks.
        \item Sorting the candidate subset $\mathcal{U}_{\text{feas}}$ takes $\mathcal{O}(|U| \log |U|)$ in the worst case, or $\mathcal{O}(|U|)$ using a single linear scan to find the minimum.
        \item Updating unit state takes $\mathcal{O}(1)$.
    \end{itemize}
    \item \textbf{Overall Time Complexity}: Summing across all $|T|$ trips, the total running time is:
    \begin{equation}
        \mathcal{O}\left(|T| \log |T| + |T| \cdot |U|\right)
    \end{equation}
    In typical practical railway networks where $|U| \ge \log |T|$, the asymptotic running time simplifies directly to:
    \begin{equation}
        \mathcal{O}(|T| \cdot |U|)
    \end{equation}
    \item \textbf{Space Complexity}: Maintaining unit circulation chains and state arrays requires $\mathcal{O}(|T| + |U|)$ auxiliary memory.
\end{enumerate}

\section{Why Greedy Fails: Myopic Decisions and No Look-Ahead}
Although the greedy construction heuristic is fast and deterministic, it exhibits fundamental failure modes when applied to dense, constrained rolling stock circulation networks:

\subsection{1. Myopic Local Optimization}
The greedy criterion minimizes immediate deadhead repositioning $\delta(\text{loc}_u, o_i)$ for the current trip $i$. It evaluates decisions purely in isolation, completely blind to future timetable commitments. Consequently, a unit is often dispatched to serve a minor trip merely because it is located nearby, leaving no units available for high-demand or geographically isolated trips scheduled shortly thereafter.

\subsection{2. Lack of Downstream Look-Ahead}
In railway operations, units must complete round-trip circuits returning to maintenance-equipped depots. The greedy policy has zero look-ahead capability:
\begin{itemize}
    \item Units are stranded at peripheral terminus stations without adequate turnaround windows for scheduled return revenue trips.
    \item When peak evening return trips begin, the fleet is geographically misallocated across the network, forcing massive, expensive deadhead transits or catastrophic schedule delays.
\end{itemize}

\subsection{3. Error Cascade and Infeasibility Propagation}
Unlike metaheuristics or mathematical programming, the greedy heuristic cannot backtrack or undo an assignment once made:
\begin{equation}
    \text{Suboptimal assignment at } t_1 \implies \text{Severe resource deficit at } t_2 \implies \text{Forced timing violations at } t_3
\end{equation}
As demonstrated in empirical network scaling experiments, once trip density exceeds network capacity, the greedy algorithm cascades into hundreds of minutes of timing violations and heavy unmet passenger demand.
